2. Add pre-query logic for status filters (AFTER subscriptions query, BEFORE main entries query):

```typescript
// Track entry IDs to filter by status (null means no status filter)
let statusFilteredIds: string[] | null = null;

// Handle is_starred filter
if (isStarredParam === 'true') {
  const { data: starred, error: starredError } = await supabase
    .from('user_entry_status')
    .select('entry_id')
    .eq('user_id', userId)
    .eq('is_starred', true);

  if (starredError) {
    console.error('Starred query error:', starredError);
    return serverError();
  }

  statusFilteredIds = starred?.map(s => s.entry_id) || [];

  // If no starred entries, return empty result immediately
  if (statusFilteredIds.length === 0) {
    return paginatedResponse([], { next_cursor: null, has_more: false, limit });
  }
}

// Handle is_read filter
if (isReadParam === 'true') {
  const { data: read, error: readError } = await supabase
    .from('user_entry_status')
    .select('entry_id')
    .eq('user_id', userId)
    .eq('is_read', true);

  if (readError) {
    console.error('Read query error:', readError);
    return serverError();
  }

  const readIds = read?.map(s => s.entry_id) || [];

  // If combined with starred filter, intersect the sets
  if (statusFilteredIds !== null) {
    const readSet = new Set(readIds);
    statusFilteredIds = statusFilteredIds.filter(id => readSet.has(id));
  } else {
    statusFilteredIds = readIds;
  }

  if (statusFilteredIds.length === 0) {
    return paginatedResponse([], { next_cursor: null, has_more: false, limit });
  }
}

// Handle is_read=false (unread entries)
if (isReadParam === 'false') {
  const { data: read, error: readError } = await supabase
    .from('user_entry_status')
    .select('entry_id')
    .eq('user_id', userId)
    .eq('is_read', true);

  if (readError) {
    console.error('Read query error:', readError);
    return serverError();
  }

  // For unread, we need to EXCLUDE these IDs
  // Store separately and handle in query
  const readIds = read?.map(s => s.entry_id) || [];

  // If there are read entries to exclude, we'll handle in the main query
  // by adding .not('id', 'in', readIds) - but empty array causes issues
  // So we track this separately
  if (readIds.length > 0) {
    // Use a different approach: we'll add the NOT IN to the main query
    // But first, if we already have statusFilteredIds, filter those
    if (statusFilteredIds !== null) {
      const readSet = new Set(readIds);
      statusFilteredIds = statusFilteredIds.filter(id => !readSet.has(id));
      if (statusFilteredIds.length === 0) {
        return paginatedResponse([], { next_cursor: null, has_more: false, limit });
      }
    }
  }
  // Note: If statusFilteredIds is still null after this, we need to exclude readIds in the main query
}
```

Add search functionality using ILIKE for partial matching across title, description, and content.

1. Add parameter parsing:
```typescript
const search = url.searchParams.get('search');
```

2. Add search filter to query (after other filters, before execute):
```typescript
// Apply search filter
if (search && search.trim()) {
  const searchTerm = search.trim();
  // ILIKE for case-insensitive partial match across multiple columns
  // Supabase handles escaping in parameterized queries
  query = query.or(
    `title.ilike.%${searchTerm}%,description.ilike.%${searchTerm}%,content.ilike.%${searchTerm}%`
  );
}
```

**Important notes:**
- Supabase JS client parameterizes queries, preventing SQL injection
- ILIKE is PostgreSQL case-insensitive LIKE
- The `%` wildcards enable partial matching (contains)
- Searching empty string should be treated as no search
- This approach is simpler than full-text search and doesn't require schema changes
- Performance is acceptable for moderate data sizes; if it becomes slow, can upgrade to full-text search later

3. Ensure search can be combined with all other filters (it's just another WHERE clause).
